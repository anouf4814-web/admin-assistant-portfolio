        async function syncFromGoogleCloud(silent = true) {
            try {
                const res = await fetch(CLOUD_API_URL);
                const json = await res.json();
                if (json.status === "success") {
                    if (json.settings) {
                        state.portfolio = {
                            ...DEFAULT_PORTFOLIO_DATA,
                            ...json.settings,
                            cv: { ...DEFAULT_PORTFOLIO_DATA.cv, ...(json.settings.cv || {}) },
                            goals: Array.isArray(json.settings.goals) && json.settings.goals.length === 5 ? json.settings.goals : DEFAULT_PORTFOLIO_DATA.goals
                        };
                    }
                    const mappedEvidences = (json.data || []).map(row => {
                        const secObj = SECTIONS_INDEX.find(s => s.id === row.section || s.category === row.section);
                        return {
                            id: String(row.id),
                            category: row.section || 'evidences_lib',
                            sectionTitle: row.description || (secObj ? secObj.title : 'الشواهد'),
                            title: row.title || 'شاهد',
                            fileName: row.title || 'file',
                            url: row.fileUrl || '',
                            fileType: row.fileType || 'doc',
                            fileTypeLabel: getFileTypeArabicLabel(row.fileType),
                            dateFormatted: row.date || ''
                        };
                    }).reverse();
                    state.evidences = mappedEvidences;
                    if (!silent) showToast('تم تحديث البيانات من السحابة بنجاح.', 'success');
                }
            } catch (err) {
                console.error("Cloud Sync Error:", err);
                showToast('تعذر الاتصال بالسحابة. تأكد من الإنترنت.', 'warning');
            }
        }

        function getFileTypeArabicLabel(t) {
            if (t === 'image') return 'صورة';
            if (t === 'pdf') return 'ملف PDF';
            if (t === 'video') return 'فيديو';
            return 'مستند';
        }

        window.handleOwnerAuth = async function(e) {
            e.preventDefault();
            const email = document.getElementById('authEmail').value.trim().toLowerCase();
            const password = document.getElementById('authPassword').value;
            const btn = document.getElementById('authSubmitBtn');
            
            btn.disabled = true;
            btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin"></i> <span>جاري التحقق...</span>`;

            try {
                await syncFromGoogleCloud(true);
                const savedCreds = state.portfolio.ownerCreds;

                if (!savedCreds) {
                    state.portfolio.ownerCreds = { email, password };
                    state.isOwner = true;
                    sessionStorage.setItem('portfolio_is_owner', 'true');
                    await savePortfolioUpdates({ ownerCreds: { email, password } }, true);
                    showToast('تم اعتماد بياناتك كـ مالك وحيد للملف بنجاح!', 'success');
                    closeAuthModal();
                } else if (savedCreds.email === email && savedCreds.password === password) {
                    state.isOwner = true;
                    sessionStorage.setItem('portfolio_is_owner', 'true');
                    showToast('تم تسجيل الدخول بصلاحيات الإدارة الكاملة.', 'success');

                btn.innerHTML = `<i class="fa-solid fa-right-to-bracket"></i> <span>دخول لوحة الإدارة</span>`;
            }
        };

        function fileToBase64(file) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.onload = () => resolve(reader.result.split(",")[1]);
                reader.onerror = err => reject(err);
                reader.readAsDataURL(file);
            });
        }

        window.handleProfileAssetUpload = async function(event, targetType) {
            if (!state.isOwner) return;
            const file = event.target.files[0];
            if (!file) return;

            showToast('جاري رفع الشعار إلى Google Drive...', 'info');
            try {
                const b64 = await fileToBase64(file);
                const res = await fetch(CLOUD_API_URL, {
                    method: "POST",
                    headers: { "Content-Type": "text/plain;charset=utf-8" },
                    body: JSON.stringify({
                        action: "uploadAssetOnly",
                        fileData: b64,
                        fileName: `${targetType}_${Date.now()}_${file.name}`,
                        mimeType: file.type,
                        fileType: "image"
                    })
                });
                const json = await res.json();
                if (json.status === "success" && json.fileUrl) {
                    if (targetType === 'moeLogo') await savePortfolioUpdates({ moeLogoUrl: json.fileUrl });
                    else if (targetType === 'schoolLogo') await savePortfolioUpdates({ schoolLogoUrl: json.fileUrl });
                    else if (targetType === 'cvPhoto') await savePortfolioUpdates({ cv: { ...state.portfolio.cv, photoUrl: json.fileUrl } });
                }
            } catch (err) {
                console.error(err);
                showToast('فشل رفع الصورة إلى Google Drive.', 'error');
            }
        };

        function classifyFileType(fileName, mimeType = '') {
            const ext = fileName.split('.').pop().toLowerCase();
            if (['jpg', 'jpeg', 'png', 'gif', 'webp', 'svg'].includes(ext) || mimeType.startsWith('image/')) {
                return { type: 'image', label: 'صورة', icon: 'fa-image', badgeColor: 'bg-emerald-50 text-emerald-700' };
            }
            if (ext === 'pdf' || mimeType === 'application/pdf') {
                return { type: 'pdf', label: 'ملف PDF', icon: 'fa-file-pdf', badgeColor: 'bg-red-50 text-red-600' };
            }
            if (['mp4', 'mov', 'avi', 'webm', 'mkv'].includes(ext) || mimeType.startsWith('video/')) {
                return { type: 'video', label: 'فيديو', icon: 'fa-video', badgeColor: 'bg-purple-50 text-purple-700' };
            }
            return { type: 'doc', label: 'مستند', icon: 'fa-file-lines', badgeColor: 'bg-sky-50 text-brand-ocean' };
        }

        function formatHijriDate() {
            try {
                return new Intl.DateTimeFormat('ar-SA-u-ca-islamic', { year: 'numeric', month: 'long', day: 'numeric' }).format(new Date());
            } catch (e) {
                return new Date().toLocaleDateString('ar-SA');
            }
        }

        window.handleEvidenceFilesUpload = async function(filesList, categoryId) {
            if (!state.isOwner) {
                showToast('إضافة الشواهد متاحة لصاحب/ة الملف فقط.', 'error');
                return;
            }
            const files = Array.from(filesList);
            if (!files.length) return;

            const customTitleInput = document.getElementById('optionalEvidenceTitle');
            const customTitle = customTitleInput ? customTitleInput.value.trim() : '';
            const progressWrapper = document.getElementById('uploadProgressWrapper');
            const progressBar = document.getElementById('uploadProgressBar');
            const progressText = document.getElementById('uploadProgressText');

            if (progressWrapper) progressWrapper.classList.remove('hidden');

            let successCount = 0;
            const sectionObj = SECTIONS_INDEX.find(s => s.id === categoryId || s.category === categoryId);
            const sectionTitle = sectionObj ? sectionObj.title : 'الشواهد';

            for (let i = 0; i < files.length; i++) {
                const file = files[i];
                const fileMeta = classifyFileType(file.name, file.type);
                const cleanBaseName = file.name.replace(/\.[^/.]+$/, "");
                const finalTitle = (files.length === 1 && customTitle) ? customTitle : (customTitle ? `${customTitle} (${i + 1})` : cleanBaseName);

                const pct = Math.round(((i + 0.5) / files.length) * 100);
                if (progressBar) progressBar.style.width = `${pct}%`;
                if (progressText) progressText.textContent = `جاري رفع (${i + 1} من ${files.length}) إلى Google Drive: ${file.name}`;

                try {
                    const b64 = await fileToBase64(file);
                    const res = await fetch(CLOUD_API_URL, {
                        method: "POST",
                        headers: { "Content-Type": "text/plain;charset=utf-8" },
                        body: JSON.stringify({
                            action: "create",
                            section: categoryId,
                            sectionTitle: sectionTitle,
                            title: finalTitle,
                            fileData: b64,
                            fileName: file.name,
                            mimeType: file.type,
                            fileType: fileMeta.type,
                            dateFormatted: formatHijriDate()
                        })
                    });
                    const json = await res.json();
                    if (json.status === "success") {
                        successCount++;
                    } else {
                        showToast(`فشل رفع ${file.name}: ${json.message || ''}`, 'error');
                    }
                } catch (err) {
                    console.error("Upload failed:", err);
                    showToast(`فشل رفع الملف: ${file.name}`, 'error');
                }
            }

            if (progressBar) progressBar.style.width = `100%`;
            if (progressWrapper) setTimeout(() => progressWrapper.classList.add('hidden'), 1000);
            if (customTitleInput) customTitleInput.value = '';

            if (successCount > 0) {
                showToast(`تم حفظ ${successCount} ${successCount === 1 ? 'شاهد' : 'شواهد'} في Google Drive بنجاح!`, 'success');
                await syncFromGoogleCloud(true);
                renderCurrentView();
            }
        };

        function computeStats() {
            const all = state.evidences;
            const imagesCount = all.filter(e => e.fileType === 'image').length;
            const pdfCount = all.filter(e => e.fileType === 'pdf').length;
            const videoCount = all.filter(e => e.fileType === 'video').length;
            const docCount = all.filter(e => e.fileType === 'doc').length;

            const initiativesCount = all.filter(e => e.category === 'initiatives').length;
            const coursesCount = all.filter(e => e.category === 'courses').length;
            const volunteeringCount = all.filter(e => e.category === 'volunteering').length;
            const certificatesCount = all.filter(e => e.category === 'certificates').length;

            const goals = state.portfolio.goals || [];
            let points = 0;
            goals.forEach(g => {
                if (g.status === 'completed') points += 100;
                else if (g.status === 'in_progress') points += 50;
            });
            const goalsPercentage = goals.length ? Math.round(points / goals.length) : 0;

            return { total: all.length, imagesCount, pdfCount, videoCount, docCount, initiativesCount, coursesCount, volunteeringCount, certificatesCount, goalsPercentage };
        }

                generateQRCode('dashboardQrCode', 92);

                case 'evidence': container.innerHTML = renderEvidenceSectionView(sec); break;

                <div class="bg-white rounded-2xl border border-gray-100 shadow-card overflow-hidden flex flex-col">
                    ${thumb}
                    <div class="p-4 flex-1 flex flex-col justify-between space-y-3">
                        <div>
                            <div class="flex justify-between text-[11px] mb-1"><span class="px-2 py-0.5 rounded ${meta.badgeColor} font-bold">${meta.label}</span><span class="text-brand-muted">${escapeHtml(ev.dateFormatted || '')}</span></div>
                            <h3 class="font-bold text-brand-navy text-sm line-clamp-2">${escapeHtml(ev.title)}</h3>
                            ${showBadge && ev.sectionTitle ? `<span class="inline-block mt-1 text-[11px] text-brand-ocean bg-brand-ice/50 px-2 py-0.5 rounded">${escapeHtml(ev.sectionTitle)}</span>` : ''}
                        </div>
                        <div class="pt-2 border-t flex items-center justify-between">
                            <div class="flex gap-1.5">
                                <button onclick="openPreviewModal('${ev.id}')" class="px-3 py-1.5 rounded-xl bg-brand-ice/70 text-brand-ocean text-xs font-bold">معاينة</button>
                                <a href="${ev.url}" target="_blank" class="px-3 py-1.5 rounded-xl bg-brand-bg text-brand-navy text-xs font-bold">فتح</a>
                            </div>
                            ${state.isOwner ? `<button onclick="confirmDeleteEvidence('${ev.id}')" class="w-8 h-8 rounded-xl bg-red-50 text-red-600 flex items-center justify-center text-xs"><i class="fa-solid fa-trash-can"></i></button>` : ''}
                        </div>
                    </div>
                </div>
            `;
        }

        window.openPreviewModal = function(id) {
            const ev = state.evidences.find(x => x.id === id);
            if (!ev) return;
            document.getElementById('previewTitle').textContent = ev.title;
            document.getElementById('previewMeta').textContent = `${ev.sectionTitle || ''} • ${ev.dateFormatted || ''}`;
            document.getElementById('previewDownloadBtn').href = ev.url;
            const body = document.getElementById('previewBody');
            body.innerHTML = ev.fileType === 'image'
                ? `<img src="${ev.url}" class="max-h-[72vh] max-w-full rounded-2xl object-contain">`
                : `<iframe src="${ev.url}" class="w-full h-[72vh] rounded-2xl border bg-white"></iframe>`;
            document.getElementById('previewModal').classList.remove('hidden');
            document.getElementById('previewModal').classList.add('flex');
        };
        window.closePreviewModal = function() { document.getElementById('previewModal').classList.add('hidden'); document.getElementById('previewBody').innerHTML = ''; };

        }
        function generateQRCode(elId, size) {
            const el = document.getElementById(elId);

            el.innerHTML = '';

        }

        function getGoalStatusBadge(s) {
            if (s === 'completed') return { label: 'مكتمل', cls: 'bg-brand-mint text-emerald-700' };
            if (s === 'in_progress') return { label: 'قيد التنفيذ', cls: 'bg-amber-50 text-amber-700' };
            return { label: 'لم يبدأ', cls: 'bg-gray-100 text-brand-muted' };
        }
        function escapeHtml(s) { return s ? String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;') : ''; }
        function showToast(msg, type = 'info') {
            const c = document.getElementById('toastContainer');
            const colors = { success: 'bg-emerald-600 text-white', error: 'bg-red-600 text-white', warning: 'bg-amber-500 text-white', info: 'bg-brand-navy text-white' };
            const t = document.createElement('div');
            t.className = `pointer-events-auto px-4 py-3 rounded-2xl shadow-lg text-xs font-bold ${colors[type] || colors.info}`;
            t.textContent = msg;
            c.appendChild(t);
            setTimeout(() => t.remove(), 4000);
        }
